// Paso 2: IIFE + Closures para Estado Orbital
const OrbitEngine = (() => {
  let state = { angle: 0, speed: 0.03, impulses: 0 };

  return {
    getAngle: () => state.angle,
    applyBoost: () => { 
      state.impulses++;
      state.speed += 0.01; 
    }
  };
})();

function mainLoop(time) {
  const dt = (time - lastTime) / 1000 || 0;
  lastTime = time;

  state.angle += state.speedFactor * dt * 60;
  const x = cx + Math.cos(state.angle) * radius;
  const y = cy + Math.sin(state.angle) * radius;
  
  requestAnimationFrame(mainLoop);
}
